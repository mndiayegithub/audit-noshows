    ```tsx
    "use client";
    import { motion, useReducedMotion } from "framer-motion";
    import { useState, type ReactNode } from "react";

    type Props = {
      children: ReactNode;
      durationSec?: number;
      ariaLabel?: string;
      className?: string;
    };

    export function Marquee({
      children,
      durationSec = 35,
      ariaLabel = "Défilement de logos",
      className = "",
    }: Props) {
      const [paused, setPaused] = useState(false);
      const prefersReduced = useReducedMotion();

      return (
        <div
          className={`relative overflow-hidden [mask-image:linear-gradient(90deg,transparent,#000_10%,#000_90%,transparent)] ${className}`}
          onMouseEnter={() => setPaused(true)}
          onMouseLeave={() => setPaused(false)}
          aria-label={ariaLabel}
        >
          <motion.div
            className="flex gap-12 w-max will-change-transform"
            animate={
              prefersReduced
                ? { x: 0 }
                : paused
                  ? undefined
                  : { x: ["0%", "-50%"] }
            }
            transition={{ duration: durationSec, repeat: Infinity, ease: "linear" }}
          >
            <div className="flex gap-12 shrink-0">{children}</div>
            <div className="flex gap-12 shrink-0" aria-hidden="true">{children}</div>
          </motion.div>
        </div>
      );
    }
    ```

    CRITICAL details (per RESEARCH §Pattern 3 + Pitfall 4):
    - Children MUST be duplicated (2 `<div>` blocks with same `{children}`) — without this, the loop shows a visible gap.
    - `x: ["0%", "-50%"]` — `-50%` (not `-100%`) because the outer container now holds 2× children.
    - `paused ? undefined : {...}` — passing `undefined` lets Framer retain position on resume (no reset flicker).
    - `aria-hidden="true"` on the duplicate (prevents screen-reader double-announcement).
    - `will-change-transform` utility class = `will-change: transform` (Pitfall 5 — GPU hint without triggering backdrop-filter repaint).
  </action>
