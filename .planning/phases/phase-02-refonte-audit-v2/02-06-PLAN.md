    ```tsx
    import { Document, Page, Text, View, StyleSheet } from "@react-pdf/renderer";
    import type { AuditStats } from "@/types/audit";
    import { computeScore, scoreBadge } from "@/lib/score";

    const COLORS = {
      bg: "#FFFFFF", ink: "#0F172A", muted: "#64748B", border: "#E5E7EB",
      primaryDark: "#064E3B",
      volumeBg: "#DFF3FF", volumeFg: "#2563EB",
      signalBg: "#DCF4E6", signalFg: "#059669",
      tauxBg: "#FCEACC", tauxFg: "#EA580C",
      argentBg: "#ECCDF8", argentFg: "#9333EA", argentDeep: "#6B21A8",
    };

    const styles = StyleSheet.create({
      page:    { backgroundColor: COLORS.bg, color: COLORS.ink, padding: 36, fontFamily: "Helvetica" },
      title:   { fontSize: 24, color: COLORS.primaryDark, marginBottom: 6, fontFamily: "Helvetica-Bold" },
      eyebrow: { fontSize: 9, color: COLORS.muted, textTransform: "uppercase", letterSpacing: 1, marginBottom: 12 },
      kpiGrid: { flexDirection: "row", flexWrap: "wrap", gap: 8, marginTop: 12 },
      kpiBox:  { width: "48%", borderRadius: 12, padding: 12 },
      kpiChip: { fontSize: 9, fontFamily: "Helvetica-Bold", marginBottom: 6 },
      kpiValue:{ fontSize: 22, fontFamily: "Helvetica-Bold" },
      kpiSub:  { fontSize: 10, color: COLORS.muted, marginTop: 2 },
      violet:  { backgroundColor: COLORS.argentDeep, color: "#FFFFFF", borderRadius: 16, padding: 20, marginTop: 16 },
      score:   { backgroundColor: COLORS.primaryDark, color: "#FFFFFF", borderRadius: 16, padding: 20, marginTop: 16 },
    });

    export default function RapportPDF({ stats, rapport, nomCabinet, periode }: { stats: AuditStats; rapport?: string; nomCabinet?: string; periode?: string }) {
      const fmt = (n: number) => n.toLocaleString("fr-FR");
      const score = computeScore(stats.taux_noshow);
      const badge = scoreBadge(score);

      return (
        <Document>
          <Page size="A4" style={styles.page}>
            <Text style={styles.eyebrow}>Audit no-shows</Text>
            <Text style={styles.title}>{nomCabinet ?? "Cabinet"}</Text>
            <Text style={{ fontSize: 10, color: COLORS.muted }}>Généré le {new Date().toLocaleDateString("fr-FR")}{periode ? ` · ${periode}` : ""}</Text>

            {/* Synthèse 4 KPIs */}
            <Text style={[styles.eyebrow, { marginTop: 24 }]}>Synthèse</Text>
            <View style={styles.kpiGrid}>
              <View style={[styles.kpiBox, { backgroundColor: COLORS.volumeBg }]}>
                <Text style={[styles.kpiChip, { color: COLORS.volumeFg }]}>Volume</Text>
                <Text style={[styles.kpiValue, { color: COLORS.volumeFg }]}>{fmt(stats.rdv_total)}</Text>
                <Text style={styles.kpiSub}>RDV analysés</Text>
              </View>
              <View style={[styles.kpiBox, { backgroundColor: COLORS.signalBg }]}>
                <Text style={[styles.kpiChip, { color: COLORS.signalFg }]}>Signal</Text>
                <Text style={[styles.kpiValue, { color: COLORS.signalFg }]}>{fmt(stats.no_shows)}</Text>
                <Text style={styles.kpiSub}>No-shows</Text>
              </View>
              <View style={[styles.kpiBox, { backgroundColor: COLORS.tauxBg }]}>
                <Text style={[styles.kpiChip, { color: COLORS.tauxFg }]}>Taux</Text>
                <Text style={[styles.kpiValue, { color: COLORS.tauxFg }]}>{stats.taux_noshow.toLocaleString("fr-FR", { maximumFractionDigits: 1 })} %</Text>
                <Text style={styles.kpiSub}>Taux de no-show</Text>
              </View>
              <View style={[styles.kpiBox, { backgroundColor: COLORS.argentBg }]}>
                <Text style={[styles.kpiChip, { color: COLORS.argentFg }]}>Argent</Text>
                {/* stats.ca_perdu VERBATIM — never multiply */}
                <Text style={[styles.kpiValue, { color: COLORS.argentFg }]}>{fmt(stats.ca_perdu)} €</Text>
                <Text style={styles.kpiSub}>CA perdu estimé</Text>
              </View>
            </View>

            {/* Manque à gagner violet */}
            <View style={styles.violet}>
              <Text style={{ fontSize: 9, color: "#FFFFFFCC", textTransform: "uppercase" }}>CA perdu / an — extrapolé</Text>
              <Text style={{ fontSize: 40, fontFamily: "Helvetica-Bold", marginTop: 4 }}>{fmt(stats.ca_perdu)} €</Text>
              <Text style={{ fontSize: 11, color: "#FFFFFFCC", marginTop: 8 }}>Valeur annualisée — extrapolée depuis la période analysée.</Text>
            </View>

            {/* Score */}
            <View style={styles.score}>
              <Text style={{ fontSize: 9, color: "#a7f3d0", textTransform: "uppercase" }}>Score cabinet</Text>
              <Text style={{ fontSize: 40, fontFamily: "Helvetica-Bold", marginTop: 4 }}>{score} / 100</Text>
              <Text style={{ fontSize: 11, color: "#d1fae5", marginTop: 6 }}>{badge.label}</Text>
            </View>

            {/* Plan */}
            <Text style={[styles.eyebrow, { marginTop: 20 }]}>Plan d&apos;action</Text>
            <View style={{ marginTop: 6 }}>
              <Text style={{ fontSize: 11 }}><Text style={{ color: COLORS.volumeFg }}>●</Text> Rappels SMS J-2 — Mois 1 · Volume</Text>
              <Text style={{ fontSize: 11, marginTop: 4 }}><Text style={{ color: COLORS.signalFg }}>●</Text> Caution créneaux sensibles — Mois 1–2 · Signal</Text>
              <Text style={{ fontSize: 11, marginTop: 4 }}><Text style={{ color: COLORS.tauxFg }}>●</Text> Liste d&apos;attente temps réel — Mois 2 · Taux</Text>
            </View>

            {rapport && (
              <View style={{ marginTop: 18 }}>
                <Text style={styles.eyebrow}>Rapport détaillé</Text>
                <Text style={{ fontSize: 10, color: COLORS.ink, lineHeight: 1.5 }}>{rapport}</Text>
              </View>
            )}

            <Text style={{ position: "absolute", bottom: 24, left: 36, right: 36, fontSize: 9, color: COLORS.muted, textAlign: "center" }}>
              Généré par GetLostRevenue — {new Date().toLocaleDateString("fr-FR")}
            </Text>
          </Page>
        </Document>
      );
    }
    ```
