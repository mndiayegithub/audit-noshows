    import type { AuditStats } from "@/types/audit";

    interface KPI {
      chip: string;
      value: number;
      sublabel: string;
      bg: string;   // pastel bg class
      fg: string;   // fg text class
      format: (n: number) => string;

    const fmtPct = (n: number) => `${n.toLocaleString("fr-FR", { maximumFractionDigits: 1 })}${nbsp}%`;
    const fmtEur = (n: number) => `${n.toLocaleString("fr-FR")}${nbsp}€`;

    export default function SyntheseKPIs({ stats }: { stats: AuditStats }) {
      const kpis: KPI[] = [
        { chip: "Volume", value: stats.rdv_total,    sublabel: "RDV analysés",      bg: "bg-kpiVolume", fg: "text-kpiVolume-fg", format: fmtNum },
        { chip: "Signal", value: stats.no_shows,     sublabel: "No-shows",          bg: "bg-kpiSignal", fg: "text-kpiSignal-fg", format: fmtNum },
        { chip: "Taux",   value: stats.taux_noshow,  sublabel: "Taux de no-show",   bg: "bg-kpiTaux",   fg: "text-kpiTaux-fg",   format: fmtPct },

        { chip: "Argent", value: stats.ca_perdu,     sublabel: "CA perdu estimé",   bg: "bg-kpiArgent", fg: "text-kpiArgent-fg", format: fmtEur },
      ];

      return (
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
          {kpis.map((k) => (
            <div key={k.chip} className={`rounded-2xl p-5 ${k.bg}`}>
              <div className={`mb-3 inline-block rounded-full bg-white/60 px-2.5 py-0.5 text-[11px] font-semibold uppercase tracking-wide ${k.fg}`}>
                {k.chip}
              </div>
              <div className={`font-fraunces text-[40px] leading-none ${k.fg}`}>
                <CountUpNumber end={k.value} format={k.format} />
              </div>
              <div className="mt-2 text-[13px] text-gray-700">{k.sublabel}</div>
            </div>
          ))}
        </div>
      );
    }
    ```

    import type { AuditStats } from "@/types/audit";

    const nbsp = " ";
    const fmtEur = (n: number) => `${Math.round(n).toLocaleString("fr-FR")}${nbsp}€`;

    export default function MoneyBuildCard({ stats }: { stats: AuditStats }) {
      // DO NOT re-multiply stats.ca_perdu — it is ALREADY ANNUALIZED by n8n
      const noShows = stats.no_shows;
      const caMoyen = stats.ca_moyen_par_rdv ?? 0;
      const nbMois = stats.nb_mois ?? 3;
      const pertePeriode = Math.round(noShows * caMoyen);
      const facteur = nbMois > 0 ? (12 / nbMois).toFixed(1).replace(".", ",") : "—";

      return (
        <div
          className="relative overflow-hidden rounded-[28px] bg-[#6B21A8] p-8 md:p-10 text-white"
          style={{ backgroundImage: "radial-gradient(circle at 0% 0%, rgba(236,205,248,0.25), transparent 55%)" }}
        >
          <div className="grid grid-cols-1 lg:grid-cols-2 gap-8 items-center">
            <div>
              <div className="mb-2 text-[11px] font-semibold uppercase tracking-wide text-white/80">
                CA perdu / an — extrapolé
              </div>
              <div className="font-fraunces text-[72px] md:text-[96px] leading-none">
                {fmtEur(stats.ca_perdu)}
              </div>
              <p className="mt-4 max-w-md text-[15px] text-white/85">
                Soit l&apos;équivalent de plusieurs semaines de CA qui s&apos;évaporent chaque année, faute de process anti no-show.
              </p>
            </div>

            <div className="rounded-2xl border border-white/20 bg-white/10 p-5 backdrop-blur-sm">
              <ul className="space-y-2.5 text-[14px]">
                <li className="flex items-center justify-between">
                  <span className="text-white/80">No-shows détectés ({nbMois}{nbsp}mois)</span>
                  <span className="font-fraunces">{noShows.toLocaleString("fr-FR")}</span>
                </li>
                <li className="flex items-center justify-between">
                  <span className="text-white/80">CA moyen par RDV</span>
                  <span className="font-fraunces">{fmtEur(caMoyen)}</span>
                </li>
                <li className="flex items-center justify-between">
                  <span className="text-white/80">Perte sur la période</span>
                  <span className="font-fraunces">{fmtEur(pertePeriode)}</span>
                </li>
                <li className="flex items-center justify-between">
                  <span className="text-white/80">Extrapolation 12{nbsp}mois</span>
                  <span className="font-fraunces">×{nbsp}{facteur}</span>
                </li>
                <li className="mt-3 flex items-center justify-between rounded-xl bg-white/15 px-3 py-2.5 border-t border-white/25">

                  <span className="font-fraunces text-lg">{fmtEur(stats.ca_perdu)}</span>
                </li>
              </ul>
            </div>
          </div>
        </div>
      );
    }
    ```
