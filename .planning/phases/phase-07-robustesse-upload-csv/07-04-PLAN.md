       ```tsx
       "use client";

       import * as Dialog from "@radix-ui/react-dialog";
       import { AlertTriangle } from "lucide-react";

       export interface DegradedConfirmDialogProps {
         open: boolean;
         onOpenChange: (open: boolean) => void;
         recoRate: number;        // 0..1
         ignoredCount: number;    // nombre de lignes non reconnues
         totalRows: number;       // total de lignes
         onConfirm: () => void;   // continuer en mode dégradé
         onCancel: () => void;    // changer de fichier
       }

       export default function DegradedConfirmDialog({
         open, onOpenChange, recoRate, ignoredCount, totalRows, onConfirm, onCancel,
       }: DegradedConfirmDialogProps) {
         const recoPct = Math.round(recoRate * 100);
         return (
           <Dialog.Root open={open} onOpenChange={onOpenChange}>
             <Dialog.Portal>
               <Dialog.Overlay className="fixed inset-0 bg-slate-900/40 backdrop-blur-sm z-50" />
               <Dialog.Content
                 className="fixed left-1/2 top-1/2 z-50 w-[92vw] max-w-md -translate-x-1/2 -translate-y-1/2 rounded-xl bg-white p-6 shadow-xl focus:outline-none"
                 aria-describedby="degraded-desc"
               >
                 <div className="flex items-start gap-3">
                   <div className="flex-shrink-0 rounded-full bg-amber-50 p-2">
                     <AlertTriangle className="h-5 w-5 text-amber-600" aria-hidden="true" />
                   </div>
                   <div className="flex-1">
                     <Dialog.Title className="font-serif text-lg font-semibold text-slate-900">
                       Données partiellement reconnues
                     </Dialog.Title>
                     <Dialog.Description id="degraded-desc" className="mt-2 text-sm text-slate-600">
                       Sur {totalRows} lignes, {ignoredCount} n&apos;ont pas pu être interprétées
                       ({recoPct} % reconnues). Vous pouvez continuer l&apos;audit en mode dégradé —
                       le rapport indiquera les lignes ignorées.
                     </Dialog.Description>
                   </div>
                 </div>
                 <div className="mt-6 flex flex-col-reverse gap-2 sm:flex-row sm:justify-end">
                   <button
                     type="button"
                     onClick={() => { onCancel(); onOpenChange(false); }}
                     className="rounded-lg px-4 py-2 text-sm font-medium text-slate-600 hover:bg-slate-100 focus:outline-none focus-visible:ring-2 focus-visible:ring-slate-300"
                   >
                     Changer de fichier
                   </button>
                   <button
                     type="button"
                     onClick={() => { onConfirm(); onOpenChange(false); }}
                     className="rounded-lg bg-primary px-4 py-2 text-sm font-medium text-white hover:bg-primaryDark focus:outline-none focus-visible:ring-2 focus-visible:ring-primary focus-visible:ring-offset-2"
                   >
                     Continuer en mode dégradé
                   </button>
                 </div>
               </Dialog.Content>
             </Dialog.Portal>
           </Dialog.Root>
         );
       }
       ```
    3. **Style verrouillé** (per D-03) : overlay `bg-slate-900/40 backdrop-blur-sm`, content `rounded-xl bg-white p-6 shadow-xl`, primary button `bg-primary hover:bg-primaryDark` (Tailwind tokens existants — NE PAS hardcoder `#064E3B` dans la className).
    4. **Accessibilité** : Dialog.Title obligatoire (Radix), `aria-describedby` lié à Description, focus trap natif. Ne pas ajouter `role` manuel — Radix le gère.
    5. **Composant standalone** : aucun appel API, aucun state interne autre que ce que Radix gère. Tous les callbacks viennent du parent.
    6. JSX strict : `&apos;` pour les apostrophes (déjà fait ci-dessus).
