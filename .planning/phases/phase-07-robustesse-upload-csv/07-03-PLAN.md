       ```ts
       import Papa from "papaparse";
       import { DEGRADED_THRESHOLD, REJECT_THRESHOLD, MIN_RDV_VALIDES } from "./audit-thresholds";
       import type { AuditErrorCode } from "@/types/audit-errors";

       export interface CSVPreviewError {
         error_code: AuditErrorCode;
         error: string;
         details?: Record<string, unknown>;
       }

       export interface CSVPreviewSuccess {
         columns: string[];
         sampleRows: Array<Record<string, string>>;  // 3 max
         totalRows: number;
         recoRate: number;        // 0..1 — fraction de lignes avec un statut reconnu
         nbRdvValides: number;    // count des lignes avec statut reconnu (pas total)
         degraded: boolean;       // recoRate < DEGRADED_THRESHOLD ET passe les seuils reject
         willReject: boolean;     // recoRate < REJECT_THRESHOLD OU nbRdvValides < MIN_RDV_VALIDES
       }

       export type CSVPreviewResult =
         | { ok: true; preview: CSVPreviewSuccess }
         | { ok: false; error: CSVPreviewError };

       const STATUTS_RECONNUS = /^(honor[ée]|honored|present|venu[e]?|absent|no[\s-]?show|annul[ée]|cancell?ed)$/i;
       const COLONNE_DATE = /^(date|jour|date_rdv|date du rendez-vous)$/i;
       const COLONNE_STATUT = /^(statut|status|état|etat)$/i;

       export function parseCSVForPreview(text: string): CSVPreviewResult {
         if (!text || text.trim().length === 0) {
           return { ok: false, error: { error_code: "EMPTY_AFTER_PARSING", error: "CSV vide ou aucune ligne lisible." } };
         }
         const parsed = Papa.parse<Record<string, string>>(text, {
           header: true,
           skipEmptyLines: true,
           transformHeader: (h) => h.trim(),
           // papaparse auto-détecte délimiteur (',', ';', '\t')
         });
         if (parsed.errors.length > 0 && parsed.data.length === 0) {
           return { ok: false, error: { error_code: "EMPTY_AFTER_PARSING", error: "Aucune ligne lisible dans le fichier." } };
         }
         const headers = parsed.meta.fields ?? [];
         const hasDate = headers.some((h) => COLONNE_DATE.test(h));
         const hasStatut = headers.some((h) => COLONNE_STATUT.test(h));
         if (!hasDate || !hasStatut) {
           const missing: string[] = [];
           if (!hasDate) missing.push("date");
           if (!hasStatut) missing.push("statut");
           return { ok: false, error: { error_code: "MISSING_COLUMNS", error: "Colonnes obligatoires manquantes.", details: { missing } } };
         }
         const statutKey = headers.find((h) => COLONNE_STATUT.test(h))!;
         const totalRows = parsed.data.length;
         if (totalRows === 0) {
           return { ok: false, error: { error_code: "EMPTY_AFTER_PARSING", error: "CSV sans données après le header." } };
         }
         const nbRdvValides = parsed.data.filter((row) => STATUTS_RECONNUS.test((row[statutKey] ?? "").trim())).length;
         const recoRate = totalRows === 0 ? 0 : nbRdvValides / totalRows;
         return {
           ok: true,
           preview: {
             columns: headers,
             sampleRows: parsed.data.slice(0, 3) as Array<Record<string, string>>,
             totalRows,
             recoRate,
             nbRdvValides,
             degraded: recoRate < DEGRADED_THRESHOLD && recoRate >= REJECT_THRESHOLD && nbRdvValides >= MIN_RDV_VALIDES,
             willReject: recoRate < REJECT_THRESHOLD || nbRdvValides < MIN_RDV_VALIDES,
           },
         };
       }
       ```

       ```ts
       "use client";
       import { useEffect, useState } from "react";
       import { parseCSVForPreview, type CSVPreviewResult } from "@/lib/parseCSVForPreview";

       export function useCSVPreview(file: File | null): {
         status: "idle" | "loading" | "ready" | "error";
         result: CSVPreviewResult | null;
       } {
         const [status, setStatus] = useState<"idle" | "loading" | "ready" | "error">("idle");
         const [result, setResult] = useState<CSVPreviewResult | null>(null);

         useEffect(() => {
           if (!file) { setStatus("idle"); setResult(null); return; }
           setStatus("loading");
           const reader = new FileReader();
           reader.onload = (e) => {
             const text = (e.target?.result as string) ?? "";
             const r = parseCSVForPreview(text);
             setResult(r);
             setStatus(r.ok ? "ready" : "error");
           };
           reader.onerror = () => {
             setResult({ ok: false, error: { error_code: "ENCODING_ERROR", error: "Impossible de lire le fichier (encodage non supporté). Réenregistrez en UTF-8." } });
             setStatus("error");
           };
           reader.readAsText(file, "utf-8");
         }, [file]);

         return { status, result };
       }
       ```
